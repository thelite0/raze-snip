cmake_minimum_required(VERSION 3.24)
project(raze-snip VERSION 0.1.2 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
include(GNUInstallDirs)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui GuiPrivate Quick Qml DBus Concurrent)
find_package(LayerShellQt REQUIRED)
find_package(KF6GlobalAccel REQUIRED)
find_package(KF6GuiAddons 6.29 REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(Wayland REQUIRED IMPORTED_TARGET wayland-client)
pkg_get_variable(WAYLAND_PROTOCOLS_DIR wayland-protocols pkgdatadir)
find_program(WAYLAND_SCANNER wayland-scanner REQUIRED)
set(POINTER_PROTOCOL ${WAYLAND_PROTOCOLS_DIR}/unstable/pointer-constraints/pointer-constraints-unstable-v1.xml)
add_custom_command(OUTPUT pointer-constraints-client.h pointer-constraints.c
 COMMAND ${WAYLAND_SCANNER} client-header ${POINTER_PROTOCOL} pointer-constraints-client.h
 COMMAND ${WAYLAND_SCANNER} private-code ${POINTER_PROTOCOL} pointer-constraints.c
 DEPENDS ${POINTER_PROTOCOL} VERBATIM)
qt_add_executable(raze-snip src/cornerguard.cpp src/cornerguard.h ${CMAKE_CURRENT_BINARY_DIR}/pointer-constraints.c ${CMAKE_CURRENT_BINARY_DIR}/pointer-constraints-client.h src/main.cpp src/capture.cpp src/capture.h src/controller.cpp src/controller.h src/canvas.cpp src/canvas.h src/geometry.h)
qt_add_resources(raze-snip "ui" PREFIX "/" FILES qml/Overlay.qml qml/Editor.qml qml/ModeButton.qml qml/ModeIcon.qml data/icons/rectangle.svg data/icons/freeform.svg data/icons/window.svg data/icons/fullscreen.svg data/icons/close.svg data/icons/camera.svg data/icons/rectangle-selected.svg data/icons/freeform-selected.svg data/icons/window-selected.svg data/icons/fullscreen-selected.svg data/icons/close-selected.svg data/icons/camera-selected.svg data/icons/chevron-selected.svg data/icons/chevron.svg)
target_include_directories(raze-snip PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(raze-snip PRIVATE Qt6::GuiPrivate PkgConfig::Wayland Qt6::Core Qt6::Gui Qt6::Quick Qt6::Qml Qt6::DBus Qt6::Concurrent LayerShellQt::Interface KF6::GlobalAccel KF6::GuiAddons)
target_compile_definitions(raze-snip PRIVATE RAZE_SNIP_VERSION="${PROJECT_VERSION}")
target_compile_options(raze-snip PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS raze-snip RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
configure_file(data/org.raze.snip.desktop.in org.raze.snip.desktop @ONLY)
configure_file(data/org.raze.snip.service.in org.raze.snip.service @ONLY)
configure_file(data/org.raze.snip.autostart.desktop.in org.raze.snip.autostart.desktop @ONLY)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/org.raze.snip.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/org.raze.snip.service DESTINATION ${CMAKE_INSTALL_DATADIR}/dbus-1/services)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/org.raze.snip.autostart.desktop DESTINATION ${CMAKE_INSTALL_SYSCONFDIR}/xdg/autostart RENAME org.raze.snip.desktop)
include(CTest)
if(BUILD_TESTING)
 find_package(Qt6 REQUIRED COMPONENTS Test)
 qt_add_executable(geometry-test tests/geometry-test.cpp)
 target_include_directories(geometry-test PRIVATE src)
 target_link_libraries(geometry-test PRIVATE Qt6::Gui Qt6::Test)
 add_test(NAME geometry COMMAND geometry-test -platform offscreen)
endif()
